\noindent\textbf{ClassDensityPtb():}
\begin{algorithmic}
% get subgraphs
\State $valEdges \gets []$
\State $d \gets $ value $\in [0\ldots1]$
\State $ptbRate \gets $ value $\geq 0$
\State $budget \gets ptbRate \times |G.edges|$
\\
\State $C_1, \ldots, C_n \gets $nodes with class ${1, \ldots, n}$
\For {$C_i \in {C_1,\ldots, C_n}$}:
\State $G_i \gets \texttt{IndSubg}(C_i)$
% get connected components
\For {$cc \in \texttt{ConComp}(G_i)$}:
\If {$\texttt{density}(cc) \geq d$}
\State $Clique \gets \texttt{MakeClique}(cc)$
\State $\texttt{append}(Clique.edges, valEdges)$
\EndIf
\EndFor
\EndFor
\\
\While {under budget}
    \State $edge \gets \texttt{randSel}(valEdges)$
    \If {$edge \notin G}$
        \State $\texttt{addEdge}(G, edge)$
    \EndIf
\EndWhile

\end{algorithmic}
