\noindent\textbf{GeneralGreedyPtb(heuristic)}
\begin{algorithmic}
% get subgraphs
\State $valEdges \gets []$
\State $ptbRate \gets $ value $\geq 0$
\State $budget \gets ptbRate \times |G.edges|$
\\
\For {$u \in G.nodes$}
\For {$v \in G.nodes$}
\If {$u \neq v$ and $\texttt{heuristic}(u, v)$}
    \State $\texttt{append}(\{u, v\}, valEdges)$
\EndIf
\EndFor
\EndFor
\\
\While {under budget}
    \State $edge \gets \texttt{randSel}(valEdges)$
    \If {$edge \notin G}$
        \State $\texttt{addEdge}(G, edge)$
        \If {accuracy changes}
            \State $\texttt{removeEdge}(G, edge)$
        \EndIf
    \EndIf
\EndWhile

\end{algorithmic}
